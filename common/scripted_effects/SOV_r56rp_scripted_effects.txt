	if = {
		limit = {
			EST = {
				exists = no
			}
			LAT = {
				exists = no
			}
			LIT = {
				exists = no
			}
		}
		release_autonomy = {
			target = LAT
			autonomy_state = autonomy_ssr
		}
	}
}

# Giving LAT our held states in the Baltics and giving them Cores
SOV_r56rp_establish_united_baltic_ssr_LAT = {
	if = {
		limit = {
			LAT = {
				is_subject_of = ROOT
			}
		}
		every_state = {
			limit = {
				is_core_of = EST
			}
			add_core_of = LAT
		}
		every_state = {
			limit = {
				is_core_of = LIT
			}
			add_core_of = LAT
		}
		every_state = {
			limit = {
				is_core_of = LAT
				is_owned_and_controlled_by = ROOT
			}
			transfer_state_to = LAT
		}
		set_autonomy = {
			target = LAT
			autonomous_state = autonomy_ssr
			end_wars = no
			end_civil_wars = no
		}
		LAT = {
			set_cosmetic_tag = BAL_UNIFIED_communism
		}
	}
}
