scripted_diplomatic_actions = {

	demilitarize_subject_action = {

		allowed = {
			ROOT = { is_ai = no }
		}

		visible = {
			PREV = {
				AND = {
					NOT = { has_idea = subject_military_restrictions }
					exists = yes
					is_subject_of = ROOT
					is_ai = yes
				}
			}
		}

		selectable = {
			custom_trigger_tooltip = {
				tooltip = RESTRICTING_ALLOWED_TOOLTIP
				always = yes
			}	
		}

		cost = 150

		command_power = 15

		requires_acceptance = no

		icon = 1

		complete_effect = {
			PREV = {
				add_timed_idea = {
					idea = recently_restricted
					days = 20
				}
				add_ideas = subject_military_restrictions
				add_ideas = cant_train_divisions
				country_lock_all_division_template = yes
				delete_unit = {
					disband = yes
				}
			}
		}
	}

	remilitarize_subject_action = {

		allowed = {
			ROOT = { is_ai = no }
		}

		visible = {
			PREV = {
				AND = {
					has_idea = subject_military_restrictions
					exists = yes
					is_subject_of = ROOT
				}
			}
		}

		selectable = {
			IF = {
				limit = {
					PREV = {
						has_idea = recently_restricted
					}
				}
				custom_trigger_tooltip = {
					tooltip = DERESTRICTING_DENIED_TOOLTIP
					always = no
				}
			}
			else = {
				custom_trigger_tooltip = {
					tooltip = DERESTRICTING_ALLOWED_TOOLTIP
					always = yes
				}
			}
		}

		cost = 25

		command_power = 5

		requires_acceptance = no

		icon = 1

		complete_effect = {
			PREV = {
				remove_ideas = subject_military_restrictions
				remove_ideas = cant_train_divisions
				country_lock_all_division_template = no
			}
		}
	}

    rp_puppet_independence_action = {
        icon = 1
        cost = 75
        command_power = 15
        requires_acceptance = no
        show_acceptance_on_action_button = no
        send_description =  RP_PUPPET_INDEPENDENCE_ACTION_DESC
		receive_description = RP_PUPPET_INDEPENDENCE_RECIEVE_DESC

        allowed = {
			ROOT = { is_ai = no }
		}

		visible = {
			PREV = {
				AND = {
					exists = yes
					is_subject_of = ROOT
				}
			}
		}

    	can_be_sent = { always = yes }

        complete_effect = {
			ROOT = {
				end_puppet = PREV
				give_guarantee = PREV
				give_military_access = PREV
			}
			PREV = {
				give_military_access = ROOT
			}
            if = {
				limit = {
					has_idea = subject_military_restrictions
				}
				remove_ideas = subject_military_restrictions
			}
            if = {
				limit = {
					has_idea = cant_train_divisions
				}
				remove_ideas = cant_train_divisions
			}
		}
	}

	r56rp_give_1_civ = {
		icon = 1
		cost = 0
		requires_acceptance = no
		send_description =  R56RP_GIVE_1_CIV_DESC
		receive_description = R56RP_GIVE_1_CIV_RECIEVE_DESC

		allowed = {
			
		}

		visible = {
			ROOT = {
				is_ai = no
			}
		}

		selectable = {
			ROOT = {
				industrial_complex > 9
			}
			THIS = {
				any_core_state = {
					free_building_slots = {
						building = industrial_complex
						size > 1
						include_locked = yes
					}
				}
			}
		}

		complete_effect = {
			FROM = {
				random_core_state = {
					limit = {
						free_building_slots = {
							building = industrial_complex
							size > 0
							include_locked = yes
						}
					}
					add_building_construction = {
						type = industrial_complex
						level = 1
						instant_build = yes
					}
				}
			}

			ROOT = {
				random_controlled_state = {
					limit = {
						industrial_complex > 0
					}
					remove_building = {
						type = industrial_complex
						level = 1
					}
				}
			}
			
		}
	}
