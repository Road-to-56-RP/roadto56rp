GENERIC_r56rp_seek_collaboration_category = {
	GENERIC_r56rp_seek_support_from_major = {
		priority = 100

		visible = {

		}

		cost = 25

		available = {
			NOT = {
				has_country_flag = GENERIC_collaborating_with_major
			}
			FROM = {
				has_opinion = {
					target = ROOT
					value > 34
				}
			}
		}

		target_trigger = {
			FROM = {
				is_major = yes
				is_subject = no
				has_capitulated = no
				has_civil_war = no
				NOT = {
					has_war_with = ROOT
				}
			}
		}

		complete_effect = {
			FROM = {
				add_political_power = 25
				country_event = { id = GENERIC_r56rp_collaboration.1 hours = 3 }
			}
		}
	}
}
