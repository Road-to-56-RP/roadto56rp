			original_tag = LAT

		}

			NOT = {
				has_country_flag = LAT_r56rp_recently_purchased_stugs
			}
			check_variable = { ordered_stugs < 5}
		}

		cost = 25

		days_remove = 120

		complete_effect = {
			set_country_flag = LAT_r56rp_recently_purchased_stugs
			if = {
				limit = {
					has_dlc = "No Step Back"
					NOT = {
						has_tech = basic_medium_tank_chassis
					}
				}
				set_technology = { basic_medium_tank_chassis = 1 }
				if = {
					limit = {
						has_dlc = "No Step Back"
						NOT = {
							has_tech = basic_heavy_tank_chassis
						}
					}
					set_technology = { basic_heavy_tank_chassis = 1 }
				}
			}
			if = {
				limit = {
					NOT = {
						has_dlc = "No Step Back"

					}
				}
				set_technology = { basic_medium_tank = 1 }
			}
			if = {
				limit = {
					NOT = {
						has_dlc = "No Step Back"
						has_tech = basic_heavy_tank
					}
				}
				set_technology = { basic_heavy_tank = 1 }
			}
			add_to_variable = {
				var = ordered_stugs
				value = 1
			}
			if = {
				limit = {
					check_variable = { ordered_stugs < 4 }
				}

				country_event = { id = LAT_r56rp_purchase_stugs.2 hours = 24 }

				if = {
					limit = {
						has_dlc = "No Step Back"

						type = heavy_tank_artillery_chassis_1
						amount = 250
						variant_name = "Panzer Selbstfahrlafette V"
					}			
				}	
			}
		}

		remove_effect = {
			clr_country_flag = LAT_r56rp_recently_purchased_stugs
		}
	}
