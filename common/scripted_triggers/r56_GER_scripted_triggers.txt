GER_naval_interest_trigger = {
	OR = {
		AND = {
			not = { has_war_with = SOV }
			OR = {
				has_war_with = ENG
				has_war_with = USA
				has_war_with = JAP
			}
		}
		has_completed_focus = GER_plan_z
		has_completed_focus = GER_trade_interdiction
	}
}

		}

		has_wargoal_against = SOV
	}
}
